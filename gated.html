💻 Gated Content Code Pattern (How it works)

// 1. User clicks "One-Click Unlock & Download" on a protected PDF or report
downloadButton.addEventListener('click', async () => {
  try {
    // 2. Immediate UI mode checks local credentials & prompts account picker
    const cred = await navigator.credentials.get({
      password: true,
      publicKey: { challenge: docChallenge, rpId: 'example.com' },
      uiMode: 'immediate'
    });
    // 3. Authenticated! Trigger the file download immediately.
    startResourceDownload(docId, cred);
  } catch (error) {
    if (error.name === 'NotAllowedError') {
      // 4. Fallback: reveal inline lead-capture form
      showInlineLeadCaptureForm(docId);
    }
  }
});
