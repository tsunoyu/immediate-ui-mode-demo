💻 Reauthentication Code Pattern (How it works)

// 1. User clicks a sensitive action link/button (e.g., Change Address, Authorize Payment)
changeAddressButton.addEventListener('click', async () => {
  try {
    // 2. Invoke Immediate UI mode to reauthenticate the signed-in user in-place
    const cred = await navigator.credentials.get({
      password: true,
      publicKey: { challenge: stepUpChallenge, rpId: 'example.com' },
      uiMode: 'immediate'
    });
    // 3. Reauth verified! Perform sensitive operation immediately.
    await executeSensitiveAction(cred);
  } catch (error) {
    if (error.name === 'NotAllowedError') {
      // 4. User cancelled or step-up failed: show standard reauth modal
      showStepUpAuthModal();
    }
  }
});
