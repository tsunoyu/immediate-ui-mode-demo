💻 Quick Multi-Account Switcher Pattern (How it works)

// 1. User clicks "Switch Account" in the SaaS header
switchAccountBtn.addEventListener('click', async () => {
  try {
    // 2. Invoke Immediate UI mode to display the browser's native account list
    const cred = await navigator.credentials.get({
      password: true,
      publicKey: { challenge: switchChallenge, rpId: 'example.com' },
      uiMode: 'immediate'
    });
    // 3. User picked a different account! Update session and dashboard.
    switchWorkspaceProfile(cred);
  } catch (error) {
    if (error.name === 'NotAllowedError') {
      // 4. User cancelled or no accounts left: show account switcher dropdown
      openCustomAccountMenu();
    }
  }
});
