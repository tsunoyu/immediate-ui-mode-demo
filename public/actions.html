💻 Action Buttons Pattern (How it works)

When a user clicks an action button (Like, Bookmark, Wishlist), your click handler checks authentication. If logged out, calling navigator.credentials.get({ uiMode: 'immediate' }) prompts the user in the same context. Upon success, you immediately update the button state! 

likeButton.addEventListener('click', async (event) => {
  if (!currentUser) {
    try {
      // Prompt user in-place without redirecting!
      const cred = await navigator.credentials.get({
        password: true,
        publicKey: { challenge: serverChallenge, rpId: 'example.com' },
        uiMode: 'immediate'
      });
      currentUser = await verifyCredential(cred);
    } catch (err) {
      if (err.name === 'NotAllowedError') {
        // Fallback: show inline toast or standard sign-in prompt
        return showInlineAuthModal();
      }
    }
  }
  // User authenticated! Record the Like immediately.
  await sendLikeRequest(itemId);
  likeButton.classList.add('liked');
});
