💻 E-Commerce Checkout Pattern (How it works)

By invoking navigator.credentials.get({ uiMode: 'immediate' }) when the user clicks Secure Checkout, returning customers sign in dynamically on-the-fly. If NotAllowedError is thrown (guest users or empty password manager), gracefully present guest checkout options. 

document.getElementById('checkout-btn').addEventListener('click', async () => {
  try {
    // 1. Proactively offer immediate login before checkout
    const credential = await navigator.credentials.get({
      publicKey: { challenge: serverChallenge, rpId: 'example.com' },
      uiMode: 'immediate'
    });
    // 2. Returning user verified! Complete order immediately.
    completeOrder(credential);
  } catch (error) {
    if (error.name === 'NotAllowedError') {
      // 3. No local credentials found.
      // Seamlessly transition to the Guest Checkout & Sign-In web form!
      showGuestCheckoutForm();
    }
  }
});
