💻 Inline Commenting Code Pattern (How it works)

// 1. User clicks "Post Comment" while typing their feedback
postCommentButton.addEventListener('click', async () => {
  const draftText = commentInput.value;
  if (!currentUser) {
    try {
      // 2. Invoke immediate UI mode—authenticating without page reload!
      const cred = await navigator.credentials.get({
        password: true,
        publicKey: { challenge: authChallenge, rpId: 'example.com' },
        uiMode: 'immediate'
      });
      currentUser = await verifyCredential(cred);
    } catch (error) {
      if (error.name === 'NotAllowedError') {
        // 3. Fallback: prompt password form in a lightweight modal while preserving draftText!
        return showInlineCommentAuthModal(draftText);
      }
    }
  }
  // 4. Authenticated! Post comment immediately with zero text loss.
  appendCommentToThread(currentUser, draftText);
});
